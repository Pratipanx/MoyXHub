local function equal(a: any,b: any): boolean
    if type(a) ~= type(b) then return false end
    if type(a) ~= "table" then return a == b end
    for k,v in pairs(a) do if not equal(v,b[k]) then return false end end
    for k in pairs(b) do if a[k]==nil then return false end end
    return true
end
local function classify(item: any): (boolean?,string)
    local model = lab.Models[item.Model]
    local spec = item.Spec
    if not model or not spec or type(spec.Features)~="table" then return nil,"ข้อมูลไม่ครบ" end
    local f = spec.Features
    local ref = lab.Genuine(item.Model).Features
    if model.Mesh then
        if type(f.Variant)~="string" then return nil,"ไม่พบพิมพ์" end
        if f.Variant ~= ref.Variant then return false,"พิมพ์ผิด: "..f.Variant end
        if f.Proc ~= nil and not equal(f.Proc,ref.Proc) then return false,"ตำหนิผิว/พิมพ์ผิด" end
    end
    for k,v in pairs(ref) do
        if f[k]==nil then return nil,"ขาดข้อมูล "..tostring(k) end
        if not equal(f[k],v) then return false,"ตำหนิผิด: "..tostring(k) end
    end
    for k in pairs(f) do
        if ref[k]==nil then return nil,"พบข้อมูลพิมพ์เพิ่มเติม "..tostring(k) end
    end
    return true,"ตรงกับพิมพ์แท้"
end

local function waitFor(t: number): boolean
    local start = os.clock()
    repeat task.wait(0.1) until not active() or os.clock()-start >= t

local function action(name: string, arg: any?): boolean
    local isUpgrade = name=="upgrade" or name=="buyUp"
    local isExtra = name=="hire" or name=="digBuy" or name=="digEquip" or name=="digRepair"
    if isUpgrade then
        if not upgradeActive() then return false end
    elseif isExtra then
        if not extraActive() then return false end
    elseif not active() then return false end
    attempts += 1
    lastAction = name
    ctl.Attempts = attempts
    ctl.LastAction = name
    local ok, yes, why = pcall(function() return Act:InvokeServer(name,arg) end)
    if ok and yes then
        accepted += 1
        ctl.ActionOK[name]=(ctl.ActionOK[name] or 0)+1
        ctl.Accepted = accepted
        ctl.LastError = ""
        print("[Moy Auto] OK",name)
        if isUpgrade or isExtra then task.wait(0.8) else waitFor(0.8) end
        return true
    end
    ctl.ActionRejected[name]=(ctl.ActionRejected[name] or 0)+1
    lastError = tostring(ok and why or yes)
    ctl.LastError = lastError
    warn("[Moy Auto] Rejected",name,lastError)
    if isUpgrade or isExtra then task.wait(3) else waitFor(3) end
    return false
end

local Pathfinding=game:GetService("PathfindingService")
local function walkNearImpl(point:Vector3,reach:number,enabled:()->boolean):boolean
    if not enabled() then return false end
    local ch=P.Character
    local hum=ch and ch:FindFirstChildOfClass("Humanoid")
    local rt=ch and ch:FindFirstChild("HumanoidRootPart")
    if not ch or not hum or not rt or not rt:IsA("BasePart") or hum.Health<=0 then return false end
    local root=rt :: BasePart
    local function valid():boolean
        return enabled() and P.Character==ch and root.Parent~=nil and hum.Health>0
    end
    local function stop()
        if root.Parent and hum.Parent then hum:MoveTo(root.Position);hum:Move(Vector3.zero) end
    end
    if hum.Sit then
        hum.Sit=false
        local deadline=os.clock()+1.5
        repeat task.wait(.1) until not valid() or not hum.Sit or os.clock()>=deadline
        if not valid() or hum.Sit then return false end
    end
    local targetRange=math.max(2,math.min(reach-0.7,6.5))
    if (root.Position-point).Magnitude<=targetRange then return true end
    local ray=RaycastParams.new();ray.FilterType=Enum.RaycastFilterType.Exclude;ray.FilterDescendantsInstances={ch};ray.RespectCanCollide=true
    local overlap=OverlapParams.new();overlap.FilterType=Enum.RaycastFilterType.Exclude;overlap.FilterDescendantsInstances={ch};overlap.RespectCanCollide=true
    local height=hum.HipHeight+root.Size.Y/2
    if hum.RigType==Enum.HumanoidRigType.R6 then local leg=ch:FindFirstChild("Left Leg");if leg and leg:IsA("BasePart") then height+=leg.Size.Y end end
    -- Direct movement first. Only use pathfinding if the direct trip cannot finish.
    if cfg.DirectWalk and valid() then
        local begin=root.Position
        local saved:{[BasePart]:boolean}={}
        local connection:RBXScriptConnection?=nil
        local ok=pcall(function()
            connection=game:GetService("RunService").Stepped:Connect(function()
                if not valid() or not cfg.DirectWalk then return end
                for _,v in ipairs(ch:GetDescendants()) do
                    if v:IsA("BasePart") and v.CanCollide then
                        if saved[v]==nil then saved[v]=true end
                        v.CanCollide=false
                    end
                end
            end)
            local timeout=os.clock()+math.clamp((begin-point).Magnitude/math.max(hum.WalkSpeed,1)*1.6+3,3,60)
            local nextMove=0
            local progressAt=os.clock()
            local progress=root.Position
            repeat
                if os.clock()>=nextMove then hum:MoveTo(point);nextMove=os.clock()+4 end
                task.wait(.08)
                if (root.Position-progress).Magnitude>=1 then progress=root.Position;progressAt=os.clock() end
                if (root.Position-point).Magnitude<=targetRange then
                    local blocked=false
                    for _,ob in ipairs(workspace:GetPartBoundsInBox(CFrame.new(root.Position),Vector3.new(2.2,3.8,2.2),overlap)) do
                        if ob.CanCollide then blocked=true;break end
                    end
                    if not blocked then break end
                end
            until not valid() or not cfg.DirectWalk or os.clock()>timeout or os.clock()-progressAt>2.2 or root.Position.Y<math.min(begin.Y,point.Y)-5
        end)
        if connection then connection:Disconnect() end
        for part,value in pairs(saved) do if part.Parent then part.CanCollide=value end end
        stop()
        task.wait(.2)
        if ok and valid() and (root.Position-point).Magnitude<=targetRange then
            ctl.DirectWalkCount=(ctl.DirectWalkCount or 0)+1
            return true
        end
        ctl.DirectWalkFailed=(ctl.DirectWalkFailed or 0)+1
        if not valid() then return false end
    end
    -- Leaving a seat can place the avatar above the walkable floor. Walk off it normally first.
    local under=workspace:Raycast(root.Position,-Vector3.yAxis*(height+1),ray)
    if under and (under.Instance:IsA("Seat") or under.Instance.Name=="Seat") then
        local exits:{{pos:Vector3,score:number}}={}
        for i=0,7 do
            local angle=i*math.pi/4
            local dir=Vector3.new(math.cos(angle),0,math.sin(angle))
            local x=root.Position+dir*5
            local hit=workspace:Raycast(x+Vector3.yAxis*2,-Vector3.yAxis*14,ray)
            if hit and hit.Normal.Y>.85 and hit.Instance.Name:lower():find("floor",1,true) and not workspace:Raycast(root.Position,dir*5,ray) then
                local pos=hit.Position+Vector3.yAxis*.1
                local standing=hit.Position+Vector3.yAxis*(height+.08)
                local blocked=false
                for _,ob in ipairs(workspace:GetPartBoundsInBox(CFrame.new(hit.Position+Vector3.yAxis*(height+.35)),Vector3.new(2.6,math.max(4.5,height*2-.5),2.6),overlap)) do
                    if ob~=hit.Instance then blocked=true;break end
                end
                if not blocked then table.insert(exits,{pos=pos,score=(standing-point).Magnitude}) end
            end
        end
        table.sort(exits,function(a,b) return a.score<b.score end)
        if exits[1] then
            hum:MoveTo(exits[1].pos)
            local t=os.clock()+2.5
            repeat task.wait(.1) until not valid() or os.clock()>t or ((root.Position-exits[1].pos)*Vector3.new(1,0,1)).Magnitude<1.5
            stop()
            if not valid() then return false end
            task.wait(.2)
        end
    end
    local candidates:{ {pos:Vector3,score:number} }={}
    for _,radius in ipairs({3.5,4.5,5.5}) do
        for i=0,7 do
            local angle=i*math.pi/4
            local x=point+Vector3.new(math.cos(angle)*radius,0,math.sin(angle)*radius)
            local hit=workspace:Raycast(x+Vector3.yAxis*5,-Vector3.yAxis*16,ray)
            if hit and hit.Normal.Y>.85 then
                local standing=hit.Position+Vector3.yAxis*(height+.08)
                if (standing-point).Magnitude<=targetRange then
                    local blocked=false
                    for _,ob in ipairs(workspace:GetPartBoundsInBox(CFrame.new(hit.Position+Vector3.yAxis*(height+.35)),Vector3.new(2.6,math.max(4.5,height*2-.5),2.6),overlap)) do
                        if ob~=hit.Instance then blocked=true;break end
                    end
                    if not blocked then table.insert(candidates,{pos=hit.Position+Vector3.yAxis*.1,score=(standing-root.Position).Magnitude}) end
                end
            end
        end
    end
    table.sort(candidates,function(a,b) return a.score<b.score end)
    local deadline=os.clock()+math.clamp((root.Position-point).Magnitude/math.max(hum.WalkSpeed,1)*3+15,18,90)
    for i=1,math.min(#candidates,10) do
        if not valid() or os.clock()>deadline then break end
        local path=Pathfinding:CreatePath({AgentRadius=2,AgentHeight=5,AgentCanJump=true,WaypointSpacing=4})
        local ok=pcall(function() path:ComputeAsync(root.Position,candidates[i].pos) end)
        if ok and path.Status==Enum.PathStatus.Success then
            local waypoints=path:GetWaypoints()
            local blocked=false
            local index=1
            local conn=path.Blocked:Connect(function(n) if n>=index then blocked=true end end)
            for j,wp in ipairs(waypoints) do
                index=j
                if not valid() or blocked or os.clock()>deadline then break end
                if (root.Position-point).Magnitude<=targetRange then break end
                if wp.Action==Enum.PathWaypointAction.Jump then hum.Jump=true end
                hum:MoveTo(wp.Position)
                local t=os.clock()+math.clamp((root.Position-wp.Position).Magnitude/math.max(hum.WalkSpeed,1)*2+1,2,7)
                repeat
                    task.wait(.1)
                    local d=root.Position-wp.Position
                    if Vector3.new(d.X,0,d.Z).Magnitude<=2 and math.abs(d.Y)<=height+3 then break end
                until not valid() or blocked or os.clock()>t or os.clock()>deadline or (root.Position-point).Magnitude<=targetRange
                local d=root.Position-wp.Position
                if Vector3.new(d.X,0,d.Z).Magnitude>2.5 and (root.Position-point).Magnitude>targetRange then break end
            end
            conn:Disconnect()
            path:Destroy()
            if valid() and (root.Position-point).Magnitude<=targetRange then
                stop();ctl.WalkCount=(ctl.WalkCount or 0)+1;return true
            end
        else
            path:Destroy()
        end
        stop()
    end
    stop()
    return false
end
-- Temporary auto-travel speed; restore the player's original speed after every trip.
local AUTO_WALK_SPEED=28
local function walkNear(point:Vector3,reach:number,enabled:()->boolean):boolean
    if not enabled() then return false end
    local ch=P.Character
    local hum=ch and ch:FindFirstChildOfClass("Humanoid")
    local root=ch and ch:FindFirstChild("HumanoidRootPart")
    if not hum or not root or not root:IsA("BasePart") or hum.Health<=0 then return false end
    if (root.Position-point).Magnitude<=math.max(2,math.min(reach-.7,6.5)) and not hum.Sit then return true end
    local oldSpeed=hum.WalkSpeed
    hum.WalkSpeed=AUTO_WALK_SPEED
    ctl.LastAutoWalkSpeed=hum.WalkSpeed
    local ok,result=pcall(walkNearImpl,point,reach,enabled)
    if hum.Parent then hum.WalkSpeed=oldSpeed end
    if not ok then
        if root.Parent and hum.Parent then hum:MoveTo(root.Position);hum:Move(Vector3.zero) end
        ctl.LastError=tostring(result)
        return false
    end
    return result==true
end

local function sit(): boolean
    local h = P.Character and P.Character:FindFirstChildOfClass("Humanoid")
    if h and h.SeatPart == seat then return true end
    if not near(seat) then report("Cannot reach chair"); return false end
    local pr = seat:FindFirstChild("SitPrompt",true)
    if not pr then pr = counter.Chair:FindFirstChild("SitPrompt",true) end
    if pr and pr:IsA("ProximityPrompt") and pr.Enabled then
        pr:InputHoldBegin()
        local held=waitFor(math.max(.1,pr.HoldDuration)+.2)
        pcall(function() pr:InputHoldEnd() end)
        if not held then return false end
    elseif h then
        h:MoveTo(seat.Position)
    end
    waitFor(1)

end
local function idsWhere(items: any, pred: (any)->boolean): {any}
    local ids = {}

end

local function upgradePlan(s: any): (string?,any?,string)
    if not s or (s.Tutorial and s.Tutorial.Active) then return nil,nil,"รอผ่านบทสอน" end
    local budget = math.max(0,(s.Cash or 0)-cfg.Reserve)
    local nextStage = s.Next
    local stageReady = cfg.AutoExpand and nextStage and not nextStage.Blocked and (s.Level or 0)>= (nextStage.Level or math.huge)
    local function gearPlan(): (string?,any?,string)
        if not cfg.AutoGear then return nil,nil,"ปิดอัปเกรดอุปกรณ์" end
        for _,id in ipairs(gearOrder) do
            local u = shopConfig.UpgradeById[id]
            if u and Hub.Flags["Gear_"..id] then
                local level = (s.Up or {})[id] or 0
                if level < shopConfig.UpgradeCap(u,s.Stage) then
                    local cost = shopConfig.UpgradePrice(u,level)
                    if cost<=budget then return "buyUp",id,u.Name.." ฿"..tostring(cost) end
                end
            end
        end
        return nil,nil,"อุปกรณ์เต็มหรือเงินไม่พอ"
    end
    if cfg.UpgradePriority=="อุปกรณ์ก่อน" then
        local verb,arg,desc=gearPlan()
        if verb then return verb,arg,desc end
    end
    if stageReady then
        if nextStage.Cost<=budget then return "upgrade",nil,nextStage.Name.." ฿"..tostring(nextStage.Cost) end
        if cfg.UpgradePriority=="ขยายร้านก่อน" then
            return nil,nil,"เก็บเงินขยายร้าน: ขาด ฿"..tostring(math.max(0,nextStage.Cost-budget))
        end
    end
    local verb,arg,desc=gearPlan()
    if verb then return verb,arg,desc end
    if cfg.AutoExpand then
        if not nextStage then return nil,nil,"ร้านขั้นสูงสุดแล้ว" end
        if nextStage.Blocked then return nil,nil,"ยังไม่ผ่านเงื่อนไขตู้โชว์" end
        if not stageReady then return nil,nil,"เลเวลยังไม่ถึง" end
        return nil,nil,"เก็บเงินขยายร้าน"
    end
    return nil,nil,desc
end

    local verb,arg,desc=upgradePlan(s)

        UI:Notify({Title="อัปเกรดสำเร็จ",Text=desc,Duration=4})
        upgradeCooldown=os.clock()+5
    end
    return true
end

local function tick()
    local s = data
    if not s then report("Waiting for state"); return end
    if s.Tutorial and s.Tutorial.Active then
        report("Tutorial "..tostring(s.Tutorial.Step).."/"..tostring(s.Tutorial.Total)..": "..tostring(s.Tutorial.Text))
        return
    end
    if not s.Open then
        report("ร้านปิด")
        if cfg.Open then action("setOpen",true) end
        return
    end
    local carry = s.Carry and s.Carry.Items or {}
    local raw = idsWhere(carry,function(v) return not v.Done end)
    local ready = idsWhere(s.Scans,function(v) return v.Done end)
    local cert = idsWhere(carry,function(v) return v.Done and v.Kind=="cert" end)
    local sell = idsWhere(carry,function(v)
        return cfg.Sell and v.Done and v.Kind=="buy" and v.Genuine==true and not v.Lock and not rare(v)
    end)
    local store = idsWhere(carry,function(v)
        return v.Done and v.Kind=="buy" and (v.Genuine~=true or not cfg.Sell or rare(v)) and not v.Lock
    end)
    if cfg.Cert and #cert > 0 then report("Return shelf"); atStation(shelf,"toShelf",cert); return end
    if #sell > 0 then report("List live"); atStation(live,"listLive",sell); return end
    if cfg.Store and #store > 0 and #(s.Safe or {}) < (s.SafeCap or 20) then report("Store scanned items"); atStation(safe,"toSafe",store); return end
    if cfg.Scan and #raw > 0 and #(s.Scans or {}) < (s.Slots or 1) then report("Load scanner"); atStation(scanner,"loadScanner",raw); return end
    if cfg.Scan and #ready > 0 and #carry < (s.CarryMax or 6) then report("Collect scanner"); atStation(scanner,"collectScanner",ready); return end
    if cfg.Scan and #(s.Pending or {}) > 0 and #carry < (s.CarryMax or 6) then report("Take pending tray"); prompt(tray,"TakePrompt"); return end
    if #carry >= (s.CarryMax or 6) then report("Carry full; pause for manual review"); return end
    if not cfg.Customer then report("หยุดรับลูกค้า"); return end
    if not sit() then report("Waiting to sit at counter"); return end
    local c = data and data.Customer
    if not c then report("Waiting for customer"); return end
    for i,v in ipairs(c.Items or {}) do
        if v.Stage == "offer" or v.Stage == "return" or v.Stage == "post" then
            local intent = c.Intent
            local verdict: boolean?, reason: string = nil,"ปิดระบบแยกพระ"

            local affordable = cfg.Buy and verdict==true and tonumber(ask) and ask<=cfg.MaxBuy and (s.Cash or 0)-ask>=cfg.Reserve
            if v.Stage == "return" then
                if not cfg.Cert then report("รอเปิดระบบคืนใบเซอร์"); return end
                verb="handBack"
            elseif v.Stage == "offer" and intent ~= "sell" then
                if canSelf then verb="self" elseif cfg.Scan then verb="machine" else report("รอเปิดระบบเครื่องตรวจ"); return end
            elseif affordable then verb=if v.Stage=="post" then "buyPost" else "buy"
            else verb="pass" end
            print("[Moy Eye]",v.Model,word,reason,"book",know,bookNeed,"->",verb)
            report("Inspect "..tostring(v.Model).." | "..word.." -> "..verb)
            Look:FireServer(i)
            local t = os.clock()
            if not waitFor(cfg.FastLook and 3.1 or 3.25) then return end
            local now = data and data.Customer
            if not now or now.Id~=c.Id or not now.Items[i] or now.Items[i].Stage~=v.Stage then return end

                if latest and latest.Id==c.Id and latest.Items[i] and latest.Items[i].Stage=="offer" then
                    action("machine",{i=i,dt=os.clock()-t})
                end
            end
            return
        end
    end
    report("Customer finished")
    action("dismiss")
end
